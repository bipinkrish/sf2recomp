cmake_minimum_required(VERSION 3.16)
project(Sf2InputTools LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
get_filename_component(SF2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SF2_SOURCES "${SF2_ROOT}/tools/third_party" CACHE PATH "Verified pinned extraction sources")
set(HAC "${SF2_SOURCES}/hactool")
set(NST "${SF2_SOURCES}/nstool")
add_compile_definitions(_FILE_OFFSET_BITS=64)

# Compile the two pinned mbedTLS versions independently, matching each upstream
# tool's source tree. Keep all objects in the target-specific build directory.
file(GLOB hac_crypto CONFIGURE_DEPENDS "${HAC}/mbedtls/library/*.c")
add_library(sf2_hac_crypto STATIC ${hac_crypto})
target_include_directories(sf2_hac_crypto PUBLIC "${HAC}/mbedtls/include")
file(GLOB hac_sources CONFIGURE_DEPENDS "${HAC}/*.c")
add_executable(hactool ${hac_sources})
target_compile_definitions(hactool PRIVATE _BSD_SOURCE _POSIX_SOURCE _POSIX_C_SOURCE=200112L _DEFAULT_SOURCE __USE_MINGW_ANSI_STDIO=1)
target_link_libraries(hactool PRIVATE sf2_hac_crypto)

function(nst_library name folder)
    file(GLOB_RECURSE sources CONFIGURE_DEPENDS "${NST}/deps/${folder}/src/*.c" "${NST}/deps/${folder}/src/*.cc" "${NST}/deps/${folder}/src/*.cpp")
    add_library(${name} STATIC ${sources})
    target_include_directories(${name} PUBLIC "${NST}/deps/${folder}/include")
endfunction()
nst_library(sf2_fmt libfmt)
nst_library(sf2_lz4 liblz4)
nst_library(sf2_nst_crypto libmbedtls)
nst_library(sf2_toolchain libtoolchain)
nst_library(sf2_pietendo libpietendo)
target_link_libraries(sf2_toolchain PUBLIC sf2_nst_crypto sf2_fmt)
target_link_libraries(sf2_pietendo PUBLIC sf2_toolchain sf2_nst_crypto sf2_fmt)
file(GLOB nst_sources CONFIGURE_DEPENDS "${NST}/src/*.cpp")
add_executable(nstool ${nst_sources})
target_link_libraries(nstool PRIVATE sf2_pietendo sf2_toolchain sf2_fmt sf2_lz4 sf2_nst_crypto)

if(WIN32)
    if(NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Windows input tools require the native MSYS2 UCRT64 MinGW toolchain")
    endif()
    add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
    # NSTool defines wmain(), while hactool uses UTF-8 main().
    target_link_options(nstool PRIVATE -municode -static)
    target_link_options(hactool PRIVATE -static)
    target_link_libraries(sf2_hac_crypto PUBLIC ws2_32 advapi32)
    target_link_libraries(sf2_nst_crypto PUBLIC ws2_32 advapi32)
    enable_language(RC)
    configure_file(windows.rc.in "${CMAKE_BINARY_DIR}/windows.rc" @ONLY)
    target_sources(hactool PRIVATE "${CMAKE_BINARY_DIR}/windows.rc")
    target_sources(nstool PRIVATE "${CMAKE_BINARY_DIR}/windows.rc")
else()
    target_compile_definitions(sf2_toolchain PRIVATE _GNU_SOURCE)
    target_link_libraries(hactool PRIVATE m)
    target_link_libraries(nstool PRIVATE m)
endif()
